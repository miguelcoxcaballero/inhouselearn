import urllib.request, io, colorsys, sys
from PIL import Image
base = sys.argv[1]
names = sys.argv[2:]
for n in names:
    data = urllib.request.urlopen(base + n).read()
    im = Image.open(io.BytesIO(data)).convert('RGB')
    W, H = im.size
    cols = 150
    rows = max(10, int(H / W * cols * 0.5))
    px = im.load()
    print('==', n, W, 'x', H)
    lil = []
    for y in range(0, H, 2):
        for x in range(0, W, 2):
            R, G, B = px[x, y]
            h, s, v = colorsys.rgb_to_hsv(R / 255, G / 255, B / 255)
            if 0.66 <= h <= 0.81 and s > 0.12 and v > 0.15:
                lil.append((x, y, s * v))
    strong = [p for p in lil if p[2] > 0.3]
    if strong:
        xs = [p[0] for p in strong]; ys = [p[1] for p in strong]
        top = max(strong, key=lambda p: p[2])
        print('lilac px:', len(lil), 'strong:', len(strong), 'bbox strong x', min(xs), max(xs), 'y', min(ys), max(ys), 'brightest', top)
    else:
        print('lilac px:', len(lil), 'strong: 0')
    for r in range(rows):
        line = ''
        for c in range(cols):
            x0 = int(c * W / cols); x1 = max(x0 + 1, int((c + 1) * W / cols)); y0 = int(r * H / rows); y1 = max(y0 + 1, int((r + 1) * H / rows))
            best = 0; lum = 0; cnt = 0
            for y in range(y0, y1, max(1, (y1 - y0) // 4)):
                for x in range(x0, x1, max(1, (x1 - x0) // 3)):
                    R, G, B = px[x, y]
                    h, s, v = colorsys.rgb_to_hsv(R / 255, G / 255, B / 255)
                    lum += v; cnt += 1
                    if 0.66 <= h <= 0.81 and s > 0.12 and v > 0.12:
                        best = max(best, s * v)
            l = lum / cnt
            if best > 0.45: ch = '#'
            elif best > 0.3: ch = '*'
            elif best > 0.16: ch = '+'
            elif best > 0.06: ch = ':'
            elif best > 0.02: ch = '.'
            else: ch = ' ' if l < 0.1 else ('-' if l < 0.55 else ('=' if l < 0.93 else ' '))
            line += ch
        print('|' + line + '|')
