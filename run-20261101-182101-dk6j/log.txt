import urllib.request, io, colorsys
from PIL import Image
base = 'https://raw.githubusercontent.com/miguelcoxcaballero/inhouselearn/screenshots/shot-20261101-181957-orh5/'
def load(n): return Image.open(io.BytesIO(urllib.request.urlopen(base + n).read())).convert('RGB')
def violet(p):
    h, s, v = colorsys.rgb_to_hsv(*(c / 255 for c in p))
    return (0.66 <= h <= 0.81 and s > 0.12 and v > 0.12), s * v
for n in ['1-oscuro-0-3-s.png', '2-oscuro-0-75-s.png', '3-oscuro-fijo.png', '4-claro-0-5-s.png', '5-claro-fijo.png', '6-movil-oscuro-0-5-s.png']:
    im = load(n); W, H = im.size; px = im.load()
    best = (0, 0, 0)
    for y in range(0, H, 2):
        for x in range(0, W, 2):
            ok, k = violet(px[x, y])
            if ok and k > best[0]: best = (k, x, y)
    # outside glow: brightest violet pixel more than 6 px away from any border pixel row band (rough: above y=60 or below the card)
    print('==', n, W, 'x', H, 'strongest violet', round(best[0], 2), 'at', best[1:], px[best[1], best[2]])
    cols = 120; rows = max(8, int(H / W * cols * 0.5))
    for r in range(rows):
        line = ''
        for c in range(cols):
            x0 = int(c * W / cols); x1 = max(x0 + 1, int((c + 1) * W / cols)); y0 = int(r * H / rows); y1 = max(y0 + 1, int((r + 1) * H / rows))
            m = 0; lum = 0; cnt = 0
            for y in range(y0, y1, max(1, (y1 - y0) // 4)):
                for x in range(x0, x1, max(1, (x1 - x0) // 3)):
                    ok, k = violet(px[x, y]); lum += max(px[x, y]) / 255; cnt += 1
                    if ok: m = max(m, k)
            l = lum / cnt
            line += '#' if m > 0.45 else '*' if m > 0.3 else '+' if m > 0.16 else ':' if m > 0.06 else '.' if m > 0.02 else (' ' if l < 0.1 or l > 0.93 else '-')
        print('|' + line + '|')
im = load('1-oscuro-0-3-s.png'); px = im.load()
print('dark 0.3s: above the top edge at x=480:', ' '.join('%d:%02x%02x%02x' % ((y,) + px[480, y]) for y in range(69, 0, -4)))
im = load('4-claro-0-5-s.png'); px = im.load()
print('light 0.5s, top row y=70 x 300..1000:', ' '.join('%d:%02x%02x%02x' % ((x,) + px[x, 70]) for x in range(300, 1000, 25)))
