from PIL import Image
base = 'https://raw.githubusercontent.com/miguelcoxcaballero/inhouselearn/screenshots/shot-20261101-193250-teyv/'
for f, label in (('3-encendido-claro-0-65-s.png', 'LIGHT desktop 0.65s'), ('4-encendido-movil-oscuro-0-65-s.png', 'DARK phone 0.65s'), ('5-encendido-movil-oscuro-fijo.png', 'DARK phone settled')):

    print('=====', label, im.size)
    cx = W // 2
    print(' centre column, bottom 230px, every 10px (r,g,b):')
    print('  ', ' '.join('%d,%d,%d' % px[cx, y] for y in range(H - 230, H, 10)))
    # strongest violet-ish drop: rows where mean (b - g) over the width is highest (borders/glow)
    rows = sorted(((sum(max(0, px[x, y][2] - px[x, y][1]) for x in range(0, W, 4)) / (W / 4), y) for y in range(int(H * 0.5), H)), reverse=True)[:4]
    print(' rows with most violet (mean b-g, y):', [(round(a, 1), y) for a, y in rows])
    # glow above the top-most strong row: profile of (b-g) every 6px going up from it
    ytop = min(y for a, y in rows[:3]) if rows else H // 2
    print(' above that row at centre (b-g, every 6px, near->far):', [px[cx, y][2] - px[cx, y][1] for y in range(ytop, max(0, ytop - 96), -6)])
