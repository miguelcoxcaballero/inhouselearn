import urllib.request, io
from PIL import Image
base = 'https://raw.githubusercontent.com/miguelcoxcaballero/inhouselearn/screenshots/shot-20261101-184055-dtsy/'
files = ['1-antes-0-25-s.png', '2-antes-0-8-s.png', '3-antes-1-3-s.png', '4-antes-fijo.png']
CH = ' .:-=+*#%@'
for f in files:
    im = Image.open(io.BytesIO(urllib.request.urlopen(base + f).read())).convert('RGB')
    W, H = im.size
    px = im.load()
    pts = [(x, y) for y in range(int(H * 0.45), H) for x in range(W) if px[x, y][2] - px[x, y][1] > 14 and px[x, y][2] > 40]
    print('=====', f, im.size, 'violet px:', len(pts))
    if not pts:
        continue
    xs = [p[0] for p in pts]; ys = [p[1] for p in pts]
    x0, x1, y0, y1 = max(0, min(xs) - 10), min(W, max(xs) + 10), max(0, min(ys) - 10), min(H, max(ys) + 10)
    print('bbox', x0, y0, x1, y1, '| peak (r,g,b):', max((px[x, y] for x, y in pts), key=lambda c: c[2]))
    cols, rows = 110, 26
    cw, rh = (x1 - x0) / cols, (y1 - y0) / rows
    for r in range(rows):
        line = ''
        for c in range(cols):
            best = 0
            for yy in range(int(y0 + r * rh), max(int(y0 + (r + 1) * rh), int(y0 + r * rh) + 1)):
                for xx in range(int(x0 + c * cw), max(int(x0 + (c + 1) * cw), int(x0 + c * cw) + 1), 2):
                    p = px[xx, yy]; best = max(best, p[2] - p[1])
            line += CH[min(9, int(best / 14))]
        print(line)
