import urllib.request, io
from PIL import Image
base = 'https://raw.githubusercontent.com/miguelcoxcaballero/inhouselearn/screenshots/shot-20261101-180833-t0zc/'
def load(n):
    return Image.open(io.BytesIO(urllib.request.urlopen(base + n).read())).convert('RGB')
def col(im, x, y0, y1, step=3):
    px = im.load()
    out = []
    rng = range(y0, y1, step) if y1 > y0 else range(y0, y1, -step)
    for y in rng:
        out.append('%d:%s' % (y, '%02x%02x%02x' % px[x, y]))
    return ' '.join(out)
def row(im, y, x0, x1, step=8):
    px = im.load()
    return ' '.join('%d:%02x%02x%02x' % ((x,) + px[x, y]) for x in range(x0, x1, step))
im = load('1-oscuro-0-3-s.png')
print('F1 dark 0.3s: vertical profile above the top edge, behind the head (x=470) and at the head (x=486)')
print(col(im, 470, 78, 0)); print(col(im, 486, 78, 0))
print('F1 the top border row y=70 and y=72 along x (the light and its tail):')
print(row(im, 70, 300, 700, 10)); print(row(im, 72, 300, 700, 10))
print('F1 inside the card just under the top edge (y=80), any lilac over the text area?')
print(row(im, 80, 300, 700, 20))
im = load('4-oscuro-fijo.png')
print('F4 dark steady: profile below the bottom edge at x=640, and above the top edge')
print(col(im, 640, 160, 227)); print(col(im, 640, 80, 30))
print('F4 the tag row y=70 x 290..420'); print(row(im, 70, 290, 420, 4))
im = load('5-claro-0-65-s.png')
print('F5 light: background and card colours, profile below the bottom edge at x=900')
print(col(im, 900, 150, 227)); print(row(im, 176, 700, 1000, 10))
im = load('6-movil-oscuro-0-5-s.png')
print('F6 phone dark 0.5s: profile above the top edge at the strongest x (looking 500..740)')
px = im.load()
best = None
for x in range(300, 760, 4):
    for y in range(1380, 1480, 2):
        r, g, b = px[x, y]
        s = b - g
        if best is None or s > best[0]: best = (s, x, y, (r, g, b))
print('strongest violet', best)
print(col(im, best[1], best[2] + 30, best[2] - 150, 6))
