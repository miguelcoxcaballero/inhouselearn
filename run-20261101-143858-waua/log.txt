#include <ft2build.h>
#include FT_FREETYPE_H
#include FT_TRIGONOMETRY_H
#include <stdio.h>
#include <stdint.h>
static uint64_t h = 1469598103934665603ULL;
static void mix(int64_t v){ h ^= (uint64_t)v; h *= 1099511628211ULL; }
static uint64_t s = 0x9E3779B97F4A7C15ULL;
static uint64_t rnd(void){ s ^= s << 13; s ^= s >> 7; s ^= s << 17; return s; }
int main(int argc, char** argv){
  const char* f = argv[1];
  const int64_t full = 360L << 16;
  int chunk = 0;
  if (f[0]=='s' || f[0]=='t' || f[0]=='u') {
    for (int64_t a = -full; a <= full; a++) {
      if (f[0]=='s') mix(FT_Sin(a));
      else if (f[0]=='t') mix(FT_Tan(a));
      else { FT_Vector v; FT_Vector_Unit(&v, a); mix(v.x); mix(v.y); }
      if (((a + full) & 0xFFFF) == 0xFFFF) { printf("%d %016llx\n", chunk++, (unsigned long long)h); }
    }
  } else {
    for (int i = 0; i < 20000000; i++) {
      int64_t m = (i % 4 == 0) ? 0xFF : (i % 4 == 1) ? 0xFFFF : (i % 4 == 2) ? 0xFFFFFF : 0x3FFFFFFF;
      int64_t x = (int64_t)(rnd() % (2*m+1)) - m, y = (int64_t)(rnd() % (2*m+1)) - m;
      int64_t ang = (int64_t)(rnd() % (4*full)) - 2*full;
      FT_Vector v = {x, y};
      if (f[0]=='r') { FT_Vector_Rotate(&v, ang); mix(v.x); mix(v.y); }
      else if (f[0]=='l') { mix(FT_Vector_Length(&v)); }
      else if (f[0]=='p') { FT_Fixed len; FT_Angle an; FT_Vector_Polarize(&v, &len, &an); mix(len); mix(an); }
      else if (f[0]=='f') { FT_Vector o; FT_Vector_From_Polar(&o, x < 0 ? -x : x, ang); mix(o.x); mix(o.y); }
      if (i % 1000000 == 999999) printf("%d %016llx\n", chunk++, (unsigned long long)h);
    }
  }
  return 0;
}
EOF
