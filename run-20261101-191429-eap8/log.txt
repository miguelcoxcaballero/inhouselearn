import urllib.request, io
from PIL import Image
base = 'https://raw.githubusercontent.com/miguelcoxcaballero/inhouselearn/screenshots/shot-20261101-191305-48tn/'
files = ['1-0-3-s.png', '2-0-65-s.png', '3-1-0-s.png', '4-1-4-s.png', '5-1-9-s.png', '6-fijo.png']
CH = ' .:-=+*#%@'
def load(f): return Image.open(io.BytesIO(urllib.request.urlopen(base + f).read())).convert('RGB')
fixed = load(files[-1]); W, H = fixed.size; fpx = fixed.load()
# composer top border = row (in lower half) with the most violet pixels in the fixed frame
def violet(p): return p[2] - p[1]
rows = {y: sum(1 for x in range(0, W, 2) if violet(fpx[x, y]) > 14 and fpx[x, y][2] > 40) for y in range(int(H * 0.5), H)}
top = max(rows, key=rows.get)
# horizontal extent of that line
xs = [x for x in range(W) if violet(fpx[x, top]) > 14 and fpx[x, top][2] > 40]
x0, x1 = min(xs), max(xs); cx = (x0 + x1) // 2
print('composer top border y=%d x=%d..%d (w=%d)' % (top, x0, x1, x1 - x0))
for f in files:
    im = load(f); px = im.load()
    print('=====', f)
    # profile above the top border at the centre and 1/4 of the width
    for label, x in (('centre', cx), ('x@25%', x0 + (x1 - x0) // 4)):
        prof = [max(0, violet(px[x, y])) for y in range(top - 96, top - 2, 8)]
        print(' %-7s above top edge (every 8px, far→near):' % label, prof)
    ys0, ys1 = max(0, top - 80), H
    cols, nrows = 100, 24
    cw, rh = (x1 - x0 + 160) / cols, (ys1 - ys0) / nrows
    for r in range(nrows):
        line = ''
        for c in range(cols):
            best = 0
            for yy in range(int(ys0 + r * rh), max(int(ys0 + (r + 1) * rh), int(ys0 + r * rh) + 1)):
                for xx in range(int(x0 - 80 + c * cw), max(int(x0 - 80 + (c + 1) * cw), int(x0 - 80 + c * cw) + 1), 2):
                    if 0 <= xx < W: best = max(best, violet(px[xx, yy]))
            line += CH[max(0, min(9, int(best / 14)))]
        print(line)
