import re, glob, collections, sys
defs = {}
exports = []
files = glob.glob("core/libs/prx/**/*.cpp", recursive=True)
print("files", len(files), flush=True)
for f in files:
    if "/tests/" in f: continue
    txt = open(f, errors="ignore").read()
    lib = f.split("/")[3]
    for m in re.finditer(r'\b(\w+)\s*\(([^;{)]*)\)\s*(?:noexcept\s*)?\{', txt):
        name = m.group(1)
        i = m.end(); depth = 1
        while i < len(txt) and depth:
            c = txt[i]
            if c == '{': depth += 1
            elif c == '}': depth -= 1
            i += 1
        body = txt[m.end():i]
        defs[(lib, name)] = ("NotImplemented" in body, len(body), f)
    for m in re.finditer(r'APS5_EXPORT\("([^"]+)",\s*(\w+)\)', txt):
        exports.append((lib, m.group(1), m.group(2), f))
print("defs", len(defs), "exports", len(exports), flush=True)
by = collections.defaultdict(list)
for lib, nid, fn, f in exports:
    d = defs.get((lib, fn))
    if d is None: continue
    by[nid].append((lib, fn, d[0], d[1], f))
out = []
for nid, lst in by.items():
    stub = [x for x in lst if x[2]]
    real = [x for x in lst if not x[2] and x[3] > 40]
    libs_r = {x[0] for x in real}
    for s in stub:
        if real and s[0] not in libs_r:
            out.append(f"CAND {s[0]} {nid} {s[1]} <= {real[0][0]} {real[0][1]} {real[0][4]}")
print("\n".join(sorted(out)))
EOF
