const { chromium } = require('/tmp/probe/node_modules/playwright-core');

const SECT = 16;
function analyse(buf, box, scale) {
  const png = PNG.sync.read(buf);
  const pad = { x: 12, y: 16 };
  const hw = box.w / 2, hh = box.h / 2, r = box.radius || 24;
  const ring = Array.from({ length: SECT }, () => ({ n: 0, c: 0, l: 0 }));
  const glow = { n: 0, l: 0 };
  const tag = { n: 0, c: 0 };
  for (let py = 0; py < png.height; py++) for (let px = 0; px < png.width; px++) {
    const sx = px / scale - pad.x, sy = py / scale - pad.y;
    const qx = Math.abs(sx - hw) - (hw - r), qy = Math.abs(sy - hh) - (hh - r);
    const d = Math.hypot(Math.max(qx, 0), Math.max(qy, 0)) + Math.min(Math.max(qx, qy), 0) - r;
    const i = (py * png.width + px) * 4;
    const R = png.data[i], G = png.data[i + 1], B = png.data[i + 2];
    const chroma = (Math.max(R, G, B) - Math.min(R, G, B)) / 255, lum = (0.2126 * R + 0.7152 * G + 0.0722 * B) / 255;
    if (Math.abs(d) <= 1.8) {
      const ang = Math.atan2(sy - hh, sx - hw); // -pi..pi, 0 = right
      const s = Math.floor(((ang + Math.PI / 2 + 2 * Math.PI) % (2 * Math.PI)) / (2 * Math.PI) * SECT); // 0 = top, clockwise
      ring[s].n++; ring[s].c += chroma; ring[s].l += lum;
    } else if (d > 2.5 && d < 14) { glow.n++; glow.l += lum; }
    if (sx > 18 && sx < 110 && sy > -11 && sy < 9) { tag.n++; tag.c += chroma; }
  }
  return { ring: ring.map((s) => (s.n ? s.c / s.n : 0)), ringL: ring.map((s) => (s.n ? s.l / s.n : 0)), glow: glow.n ? glow.l / glow.n : 0, tag: tag.n ? tag.c / tag.n : 0 };
}
